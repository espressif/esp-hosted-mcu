# Port 适配实施计划

> **For agentic workers:** Use `superpowers:subagent-driven-development` (recommended) or `superpowers:executing-plans` to implement this plan task-by-task.

**Goal:** 实现 ESP-IDF port 适配和 Linux mock port，搭建 Unity 测试框架，使核心层在真实平台和 CI 仿真平台上都能编译和运行。

**Architecture:** ESP-IDF port 将现有 `host/port/esp/freertos/src/` 下的实现搬迁到新目录结构，Linux mock port 从零构建。两个 port 实现相同的三合约接口（OSAL + Event + Transport），核心层通过 wrapper 宏零差异调用。测试套件先覆盖 OSAL 和 Transport 层，再扩展至 RPC 和事件层。

**Tech Stack:** ESP-IDF v5.3, FreeRTOS, POSIX (Linux mock), pthread, Unity test framework

**前置依赖:** 计划 1 完成（`host/core/` 和 `host/port/include/` 就位）

**产出:** 所有 `.c` 源文件可完整链接为目标平台的可执行文件；Linux mock 上 Unity 测试套件全部通过

    va_start(args, fmt);

/* host/port/esp-idf/h_event.c
 * ESP-IDF Event port — maps h_event_contract_t to esp_event API. */

#include "h_port_contract.h"
#include <esp_event.h>

/* Map ESP-IDF error codes to portable h_err_t per doc §2.5(2) */
static h_err_t esp_err_to_h_err(esp_err_t err)
{
    switch (err) {
        case ESP_OK:                return H_OK;
        case ESP_ERR_NO_MEM:        return H_ERR_NO_MEM;
        case ESP_ERR_INVALID_ARG:   return H_ERR_INVALID_ARG;
        case ESP_ERR_TIMEOUT:       return H_ERR_TIMEOUT;
        default:                    return H_FAIL;
    }
}

static int hosted_event_register(h_event_base_t base, int32_t event_id,
                                 h_event_handler_t handler, void *user_ctx)
{
    esp_event_base_t esp_base;
    switch (base) {
        case H_EVENT_WIFI: esp_base = WIFI_EVENT; break;
        case H_EVENT_IP:   esp_base = IP_EVENT;   break;
        default: return H_ERR_INVALID_ARG;
    }
    return esp_err_to_h_err(esp_event_handler_register(
        esp_base, event_id, (esp_event_handler_t)handler, user_ctx));
}

static int hosted_event_unregister(h_event_base_t base, int32_t event_id,
                                   h_event_handler_t handler)
{
    esp_event_base_t esp_base;
    switch (base) {
        case H_EVENT_WIFI: esp_base = WIFI_EVENT; break;
        case H_EVENT_IP:   esp_base = IP_EVENT;   break;
        default: return H_ERR_INVALID_ARG;
    }
    return esp_err_to_h_err(esp_event_handler_unregister(
        esp_base, event_id, (esp_event_handler_t)handler));
}

static int hosted_event_post(h_event_base_t base, int32_t event_id,
                             void *event_data, size_t event_data_size)
{
    esp_event_base_t esp_base;
    switch (base) {
        case H_EVENT_WIFI: esp_base = WIFI_EVENT; break;
        case H_EVENT_IP:   esp_base = IP_EVENT;   break;
        default: return H_ERR_INVALID_ARG;
    }
    return esp_err_to_h_err(esp_event_post(
        esp_base, event_id, event_data, event_data_size, portMAX_DELAY));
}

/* ── Global Event Instance ── */
const h_event_contract_t g_h_event = {
    .register_handler   = hosted_event_register,
    .unregister_handler = hosted_event_unregister,
    .post               = hosted_event_post,
};

                               void (*fn)(void*), void *arg, h_thread_t *out)

    return H_OK;

/* host/port/linux/src/h_event.c */
#include "h_port_contract.h"

/* Simple linked-list event handler registry */
typedef struct event_handler_entry {
    h_event_base_t base;
    int32_t event_id;
    h_event_handler_t handler;
    void *user_ctx;
    struct event_handler_entry *next;
} event_handler_entry_t;

static event_handler_entry_t *g_handlers = NULL;

static int linux_event_register(h_event_base_t base, int32_t event_id,
                                h_event_handler_t handler, void *user_ctx)
{
    event_handler_entry_t *e = malloc(sizeof(*e));
    if (!e) return H_ERR_NO_MEM;
    e->base = base; e->event_id = event_id;
    e->handler = handler; e->user_ctx = user_ctx;
    e->next = g_handlers;
    g_handlers = e;
    return H_OK;
}

static int linux_event_unregister(h_event_base_t base, int32_t event_id,
                                  h_event_handler_t handler)
{
    event_handler_entry_t **pp = &g_handlers;
    while (*pp) {
        if ((*pp)->base == base && (*pp)->event_id == event_id &&
            (*pp)->handler == handler) {
            event_handler_entry_t *tmp = *pp;
            *pp = (*pp)->next;
            free(tmp);
            return H_OK;
        }
        pp = &(*pp)->next;
    }
    return H_ERR_INVALID_ARG;
}

static int linux_event_post(h_event_base_t base, int32_t event_id,
                            void *event_data, size_t event_data_size)
{
    int count = 0;
    for (event_handler_entry_t *e = g_handlers; e; e = e->next) {
        if (e->base == base && e->event_id == event_id) {
            e->handler(event_data, event_data_size, e->user_ctx);
            count++;
        }
    }
    return (count > 0) ? H_OK : H_OK; /* No handler is not an error */
}

const h_event_contract_t g_h_event = {
    .register_handler   = linux_event_register,
    .unregister_handler = linux_event_unregister,
    .post               = linux_event_post,
};

/* Queue implementation using pipe() + internal mutex */
typedef struct {
    int fd_read;
    int fd_write;
    uint32_t item_size;
    pthread_mutex_t lock;
} linux_queue_t;

static int linux_queue_create(uint32_t count, uint32_t item_size, h_queue_t *out)
{
    linux_queue_t *q = malloc(sizeof(*q));
    if (!q) return H_ERR_NO_MEM;
    int fds[2];
    if (pipe(fds) != 0) { free(q); return H_FAIL; }
    q->fd_read = fds[0];
    q->fd_write = fds[1];
    q->item_size = item_size;
    pthread_mutex_init(&q->lock, NULL);
    *out = (h_queue_t)q;
    (void)count;
    return H_OK;
}

static int linux_queue_send(h_queue_t q, const void *item, int32_t timeout_ms)
{
    linux_queue_t *lq = (linux_queue_t *)q;
    if (timeout_ms >= 0) {
        struct pollfd pfd = { .fd = lq->fd_write, .events = POLLOUT };
        int ret = poll(&pfd, 1, timeout_ms);
        if (ret == 0) return H_ERR_TIMEOUT;
        if (ret < 0) return H_FAIL;
    }
    pthread_mutex_lock(&lq->lock);
    ssize_t ret = write(lq->fd_write, item, lq->item_size);
    pthread_mutex_unlock(&lq->lock);
    return (ret == (ssize_t)lq->item_size) ? H_OK : H_FAIL;
}

static int linux_queue_recv(h_queue_t q, void *item, int32_t timeout_ms)
{
    linux_queue_t *lq = (linux_queue_t *)q;
    if (timeout_ms >= 0) {
        struct pollfd pfd = { .fd = lq->fd_read, .events = POLLIN };
        int ret = poll(&pfd, 1, timeout_ms);
        if (ret == 0) return H_ERR_TIMEOUT;
        if (ret < 0) return H_FAIL;
    }
    ssize_t ret = read(lq->fd_read, item, lq->item_size);
    return (ret == (ssize_t)lq->item_size) ? H_OK : H_FAIL;
}

static int linux_queue_msg_waiting(h_queue_t q)
{
    /* pipe doesn't support querying pending count — return 1 as heuristic */
    (void)q;
    return 1;
}

static int linux_queue_reset(h_queue_t q)
{
    linux_queue_t *lq = (linux_queue_t *)q;
    char buf[256];
    while (read(lq->fd_read, buf, sizeof(buf)) > 0) {
        /* drain */
    }
    return H_OK;
}

static int linux_queue_delete(h_queue_t q)
{
    linux_queue_t *lq = (linux_queue_t *)q;
    close(lq->fd_read);
    close(lq->fd_write);
    pthread_mutex_destroy(&lq->lock);
    free(lq);
    return H_OK;
}

    uint8_t req1[] = {0xAA};
    uint8_t req2[] = {0xBB};
    uint8_t rsp[256];
    uint16_t rsp_len = sizeof(rsp);

    /* Two sequential requests with different UIDs should both timeout
     * (no slave), but each should be tracked independently. */
    h_err_t r1 = h_rpc_send_request(0x10, req1, sizeof(req1),
                                    rsp, &rsp_len, 10);
    h_err_t r2 = h_rpc_send_request(0x20, req2, sizeof(req2),
                                    rsp, &rsp_len, 10);
    TEST_ASSERT_EQUAL(H_ERR_TIMEOUT, r1);
    TEST_ASSERT_EQUAL(H_ERR_TIMEOUT, r2);
}

    TEST_ASSERT_EQUAL(H_OK, ret);
    TEST_ASSERT_EQUAL(1, g_event_fired);

    ret = h_event_unregister(H_EVENT_WIFI, H_EVENT_WIFI_READY, test_handler);
    TEST_ASSERT_EQUAL(H_OK, ret);

```bash
#!/bin/bash
# scripts/check_core_isolation.sh
# CI Tier 1 — Verify host/core/ contains zero ESP-IDF dependencies.
set -euo pipefail

FORBIDDEN=""

# ESP-IDF headers (any #include "esp_*" or #include <esp_*>)
FORBIDDEN+=$(grep -rn '#include.*esp_' host/core/ \
    --include="*.c" --include="*.h" 2>/dev/null || true)

# ESP-IDF types
FORBIDDEN+=$(grep -rn 'esp_err_t\|esp_event_base_t\|esp_mac_type_t\|esp_netif_' \
    host/core/ --include="*.c" --include="*.h" 2>/dev/null || true)

# ESP-IDF log macros
FORBIDDEN+=$(grep -rn 'ESP_LOG[IEWDV]\|ESP_EARLY_LOG' host/core/ \
    --include="*.c" --include="*.h" 2>/dev/null || true)

# ESP-IDF Wi-Fi types (pre-migration)
FORBIDDEN+=$(grep -rn '\bwifi_init_config_t\b\|\bwifi_config_t\b\|\bwifi_ap_record_t\b\|\bwifi_scan_config_t\b' \
    host/core/ --include="*.c" --include="*.h" 2>/dev/null || true)

# FreeRTOS direct calls (must go through OSAL vtable)
FORBIDDEN+=$(grep -rn 'vTaskDelay\|xQueueCreate\|xQueueSend\|xQueueReceive\|xSemaphoreCreate\|xSemaphoreTake\|xSemaphoreGive\|xTaskCreate\|vTaskDelete' \
    host/core/ --include="*.c" --include="*.h" 2>/dev/null || true)

# Direct heap_caps (must use h_malloc_align)
FORBIDDEN+=$(grep -rn 'heap_caps_malloc\|heap_caps_calloc\|heap_caps_free\|heap_caps_aligned' \
    host/core/ --include="*.c" --include="*.h" 2>/dev/null || true)

# GCC constructor attribute (non-portable)
FORBIDDEN+=$(grep -rn '__attribute__\s*(\s*(\s*constructor\s*)\s*)' \
    host/core/ --include="*.c" --include="*.h" 2>/dev/null || true)

# Platform-specific includes
FORBIDDEN+=$(grep -rn '#include.*freertos/' host/core/ \
    --include="*.c" --include="*.h" 2>/dev/null || true)

if [ -n "$FORBIDDEN" ]; then
    echo "========================================="
    echo "ERROR: Non-portable dependencies found in host/core/"
    echo "========================================="
    echo "$FORBIDDEN"
    echo ""
    echo "These must be replaced with h_* equivalents or moved to host/port/<platform>/"
    exit 1
fi

echo "OK: host/core/ is ESP-IDF-free and portable"
```

#!/bin/bash
# scripts/run_linux_mock_tests.sh
# CI Tier 2 — Build and run all unit tests on Linux mock port.
set -euo pipefail

SCRIPT_DIR="$(cd "$(dirname "$0")" && pwd)"
ROOT_DIR="$(dirname "$SCRIPT_DIR")"

# ── Build test executable ──

gcc -o build/test_runner_tsan \
    -I host/core/include/h_public \
    -I host/core/include/h_internal \
    -I host/port/include \
    -I host/port/linux \
    -I common/proto \
    -I common/protobuf-c/protobuf-c \
    -I tests/unity \
    -DH_BUILD_TESTS \
    -fsanitize=thread \
    -g -O1 \
    host/core/src/*.c \
    host/port/linux/src/*.c \
    tests/test_runner.c tests/test_osal.c tests/test_event.c \
    tests/test_transport.c tests/test_rpc_core.c \
    tests/unity/unity.c \
    common/protobuf-c/protobuf-c/protobuf-c.c \
    common/proto/esp_hosted_rpc.pb-c.c \
    -lpthread

echo "=== Running tests (TSAN) ==="
TSAN_OPTIONS="second_deadlock_stack=1" ./build/test_runner_tsan -v
TSAN_EXIT=$?

# ── Report ──
echo ""
if [ "$ASAN_EXIT" = "0" ] && [ "$TSAN_EXIT" = "0" ]; then
    echo "ALL TESTS PASSED (ASAN + TSAN clean)"
    exit 0
else
    echo "TEST FAILURES DETECTED"
    exit 1
fi

```yaml
# .github/workflows/host-core-test.yml
#
# Three-tier CI for host core layer:
#   Tier 1: Compile isolation check (30s)
#   Tier 2: Linux mock tests with ASAN/TSAN (3min)
#   Tier 3: ESP-IDF slave build matrix (30min)
#
# All jobs run on ubuntu-latest with zero hardware dependencies.

name: Host Core Test

on:
  push:
    paths:
      - 'host/core/**'
      - 'host/port/include/**'
      - 'host/port/linux/**'
      - 'host/port/esp-idf/**'
      - 'common/proto/**'
      - 'common/transport/**'
      - 'common/mempool/**'
      - 'tests/**'
      - 'scripts/**'
      - '.github/workflows/host-core-test.yml'
  pull_request:
    paths:
      - 'host/core/**'
      - 'host/port/**'
      - 'common/proto/**'
      - 'tests/**'
      - 'scripts/**'

permissions:
  contents: read

jobs:

  # ── Tier 1: Compile Isolation Check (30s) ──
  isolation-check:
    runs-on: ubuntu-latest
    timeout-minutes: 2
    steps:
      - uses: actions/checkout@v4

      - name: Check core layer is ESP-IDF free
        run: bash scripts/check_core_isolation.sh

      - name: Verify core compiles with zero ESP-IDF deps
        run: |
          for f in host/core/src/*.c; do
            echo "Checking $f..."
            gcc -fsyntax-only \
              -I host/core/include/h_public \
              -I host/core/include/h_internal \
              -I host/port/include \
              -I common/proto \
              -I common/protobuf-c/protobuf-c \
              -I common/transport \
              -DH_BUILD_TESTS \
              -Wall -Werror \
              "$f"
          done

  # ── Tier 2: Linux Mock Tests (3min) ──
  linux-mock-test:
    runs-on: ubuntu-latest
    needs: isolation-check
    timeout-minutes: 5
    steps:
      - uses: actions/checkout@v4
        with:
          submodules: recursive  # Need protobuf-c

      - name: Create build directory
        run: mkdir -p build

      - name: Build test executable (ASAN + UBSAN)
        run: |
          gcc -o build/test_runner \
            -I host/core/include/h_public \
            -I host/core/include/h_internal \
            -I host/port/include \
            -I host/port/linux \
            -I common/proto \
            -I common/protobuf-c/protobuf-c \
            -I tests/unity \
            -DH_BUILD_TESTS \
            -fsanitize=address,undefined \
            -g -O0 \
            -Wall -Wextra -Werror \
            host/core/src/h_init.c \
            host/core/src/h_rpc_core.c \
            host/core/src/h_rpc_wrap.c \
            host/core/src/h_transport_drv.c \
            host/core/src/h_transport_util.c \
            host/core/src/h_serial_if.c \
            host/core/src/h_event.c \
            host/core/src/h_api.c \
            host/port/linux/src/h_osal.c \
            host/port/linux/src/h_event.c \
            host/port/linux/src/h_transport_mock.c \
            tests/test_runner.c \
            tests/test_osal.c \
            tests/test_event.c \
            tests/test_transport.c \
            tests/test_rpc_core.c \
            tests/unity/unity.c \
            common/protobuf-c/protobuf-c/protobuf-c.c \
            common/proto/esp_hosted_rpc.pb-c.c \
            -lpthread

      - name: Run tests (ASAN + UBSAN)
        run: ./build/test_runner -v

      - name: Build test executable (TSAN)
        run: |
          gcc -o build/test_runner_tsan \
            -I host/core/include/h_public \
            -I host/core/include/h_internal \
            -I host/port/include \
            -I host/port/linux \
            -I common/proto \
            -I common/protobuf-c/protobuf-c \
            -I tests/unity \
            -DH_BUILD_TESTS \
            -fsanitize=thread \
            -g -O1 \
            -Wall -Wextra -Werror \
            host/core/src/*.c \
            host/port/linux/src/*.c \
            tests/test_runner.c tests/test_osal.c tests/test_event.c \
            tests/test_transport.c tests/test_rpc_core.c \
            tests/unity/unity.c \
            common/protobuf-c/protobuf-c/protobuf-c.c \
            common/proto/esp_hosted_rpc.pb-c.c \
            -lpthread

      - name: Run tests (TSAN)
        run: TSAN_OPTIONS="second_deadlock_stack=1" ./build/test_runner_tsan -v

  # ── Tier 3: ESP-IDF Slave Build Matrix (30min) ──
  # Only compiles slave firmware; host build tested in hardware regression.
  esp-idf-build-matrix:
    runs-on: ubuntu-latest
    needs: linux-mock-test
    timeout-minutes: 45
    strategy:
      fail-fast: false
      matrix:
        target: [esp32c6]
        transport: [sdio, spi, uart]
        idf_version: [v5.3]
    steps:
      - uses: actions/checkout@v4
        with:
          submodules: recursive

      - name: Build slave (${{ matrix.target }}, ${{ matrix.transport }}, IDF ${{ matrix.idf_version }})
        uses: espressif/esp-idf-ci-action@v2
        with:
          esp_idf_version: ${{ matrix.idf_version }}
          target: ${{ matrix.target }}
          command: |
            cd slave
            idf.py -DSDKCONFIG_DEFAULTS="sdkconfig.defaults;sdkconfig.ci.${{ matrix.transport }}" build

  # ── Tier 3 (supplement): ESP-IDF Host Build Verification ──
  # Verifies ESP-IDF port files are compilable within IDF toolchain context.
  esp-idf-host-build:
    runs-on: ubuntu-latest
    needs: linux-mock-test
    timeout-minutes: 15
    strategy:
      fail-fast: false
      matrix:
        transport: [spi]
        idf_version: [v5.3]
    steps:
      - uses: actions/checkout@v4
        with:
          submodules: recursive

      - name: Verify ESP-IDF port files compile (syntax check)
        uses: espressif/esp-idf-ci-action@v2
        with:
          esp_idf_version: ${{ matrix.idf_version }}
          target: esp32p4
          command: |
            # Syntax-check ESP-IDF port source files against IDF includes
            for f in host/port/esp-idf/*.c; do
              echo "Checking $f..."
              xtensa-esp32p4-elf-gcc -fsyntax-only \
                $(idf.py --preview set-target esp32p4 2>/dev/null; echo "-I${IDF_PATH}/components/freertos/FreeRTOS-Kernel/include") \
                -I host/core/include/h_public \
                -I host/port/include \
                -I host/port/esp-idf \
                "$f" || { echo "  [ERROR] Syntax check failed — fix before merge"; exit 1; }
            done
```

# scripts/hw_smoke_test.sh
# Tier 3 — Hardware-in-the-loop layered smoke test.
# Run on the host MCU (ESP32-P4) with slave (ESP32-C6) connected via SDIO.
#
# Each step validates one abstraction layer before moving to the next.
# Do NOT skip steps — failures at lower layers make higher-layer results
# meaningless.

set -euo pipefail

# ═══════════════════════════════════════════════════════════
# BASELINE — MUST be filled in before first run.
# Record these on P4+C6 hardware before the refactor:
#   TCP: ___ Mbps  (recorded YYYY-MM-DD)
#   UDP: ___ Mbps  (recorded YYYY-MM-DD)
# ═══════════════════════════════════════════════════════════
BASELINE_TCP=___   # Mbps
BASELINE_UDP=___   # Mbps
[ "$BASELINE_TCP" = "___" ] && { echo "ERROR: Set BASELINE_TCP in this script before running"; exit 1; }
[ "$BASELINE_UDP" = "___" ] && { echo "ERROR: Set BASELINE_UDP in this script before running"; exit 1; }

RED='\033[0;31m'
GREEN='\033[0;32m'
NC='\033[0m'

step=0
pass()  { echo -e "${GREEN}[PASS]${NC} Step $step: $1"; }
fail()  { echo -e "${RED}[FAIL]${NC} Step $step: $1"; exit 1; }

echo "============================================"
echo " ESP-Hosted Hardware-in-the-Loop Smoke Test"
echo "============================================"
echo "Hardware: ESP32-P4 (host) + ESP32-C6 (slave)"

echo ""
read -p "  Did slave boot successfully? (y/n) " yn

# Step 4: RPC request/response
step=4
echo "Step $step: RPC request/response"
echo "  → Host sends h_wifi_get_mode() (simple RPC, no Wi-Fi required)"
echo "  → Verify: request UID matches response UID, return code valid"
echo ""
echo "  Run on host: hw_test_rpc"
read -p "  Did RPC round-trip succeed? (y/n) " yn
[ "$yn" = "y" ] && pass "RPC round-trip OK" || fail "RPC failed"

# Step 5: Wi-Fi control
step=5
echo "Step $step: Wi-Fi control"
echo "  → h_wifi_set_mode(AP) → h_wifi_set_config → h_wifi_start"
echo "  → Verify: external STA can scan and connect to SoftAP"
echo "  → Verify: host receives H_EVENT_WIFI_AP_STACONNECTED"
